"""Dispatch only the registered live-plan and empty-recovery siblings, in order."""
from datetime import datetime, timedelta, timezone
import fcntl
from pathlib import Path
import re
import runpy
import shlex
import subprocess

STATE = Path(__file__).resolve().parent
QUEUE = STATE / "requests"
OPS = runpy.run_path(str(STATE / "completion-writer-r2.py"))
RUN = OPS["run"]


def command(packet, field, sealed_dir):
    args = shlex.split(packet[field])
    cwd = Path(args[1])
    assert args[0] == "cd" and args[2] == "&&"
    assert cwd in (RUN.ROOT, sealed_dir)
    return cwd, args[3:]


def slot(relative, seal, predecessor, groups):
    source = RUN.RESULTS / relative / "controller-packet.json"
    packet = RUN.read_json(source)
    key = packet["id"]
    request = QUEUE / f"{key}.json"
    assert request.read_bytes() == source.read_bytes() and RUN.digest(request) == seal
    ident = packet.get("timing_reservation", key)
    minutes = packet.get("requested_timing_minutes") or packet["expected_timing_minutes"]
    canonical = QUEUE / f"{ident}.released.json"
    out = Path(packet["evidence_destination"])
    timing_cwd, timing_command = command(packet, "timing_command", source.parent)
    replay_cwd, replay_command = command(packet, "replay_command", source.parent)
    receipts = STATE / "controller-receipts" / ident
    predecessor_receipt = RUN.read_json(QUEUE / f"{predecessor}.released.json")
    assert predecessor_receipt["released_at"]

    if not canonical.exists():
        status = (QUEUE / "STATUS.md").read_text()
        assert re.findall(r"^(SETUP|TIMING|HOLD|RELEASED) ", status, re.M)[-1] == "RELEASED"
        assert not re.search(r"^TIMING " + re.escape(ident) + r" ", status, re.M)
        assert not out.exists(), f"native evidence already exists at {out}"
        OPS["probe"]()
        receipts.mkdir(parents=True)
        start = datetime.now(timezone.utc)
        end = start + timedelta(minutes=minutes)
        OPS["note"](f"TIMING {ident} {start:%Y-%m-%dT%H:%M:%SZ} {end:%Y-%m-%dT%H:%M:%SZ}\n\n"
                    f"Sealed owner retry {seal}. Groups {groups}.")
        print(f"{ident} opened for groups {groups} at {start.isoformat()}", flush=True)
        result = 1
        try:
            with (receipts / "controller.log").open("xb") as stream:
                result = subprocess.run(timing_command, cwd=timing_cwd,
                                        stdout=stream, stderr=stream).returncode
        finally:
            data = dict(id=ident, phase="timing", started=start.isoformat(), request_sha256=seal,
                        exit_status=result, groups=groups, receipt=str(out))
            try:
                OPS["probe"]()
            except (OSError, subprocess.SubprocessError) as error:
                RUN.write_json(receipts / "release-pending.json", dict(data, release_probe_error=str(error)))
                raise
            data["released_at"] = RUN.utc_now()
            RUN.write_json(canonical, data)
            RUN.write_json(receipts / "status.json", data)
            OPS["note"](f"RELEASED {ident} {data['released_at']}, exit {result}. Evidence: {out}")

    data = RUN.read_json(canonical)
    assert data["id"] == ident and data["request_sha256"] == seal and data["groups"] == groups
    out.mkdir(parents=True, exist_ok=True)
    release = out / "controller-release.json"
    if not release.exists():
        with release.open("xb") as stream:
            stream.write(canonical.read_bytes())
    assert release.read_bytes() == canonical.read_bytes()
    summary = out / "summary.json"
    if ("timing_reservation" not in packet or data["exit_status"] == 0) and not summary.exists():
        with summary.open("xb") as stream, (receipts / "replay.log").open("xb") as log:
            subprocess.run(replay_command, cwd=replay_cwd, stdout=stream, stderr=log)
    print(f"{ident} released exit {data['exit_status']}. {release}", flush=True)


if __name__ == "__main__":
    with (STATE / "owner-recovery-slots.lock").open("a") as lock:
        fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        slot("2026-11-25-dense-live-grant-plans/procfd-recovery",
             "1f5acbeef38ee53e6da91b9af424bdf7a31f73245724a50ab1368a2da82dfef2",
             "2026-11-25-dense-same-page-wal-retention-retry-groups-1-2", [1, 2])
        slot("2026-11-25-empty-recovery-native-retry",
             "129e05f14ac2170bd11d28fef0ef635b26cc6ab9402ea19995d030badaa1d4cb",
             "2026-11-25-dense-live-grant-plans-procfd-recovery-timing", [1])
